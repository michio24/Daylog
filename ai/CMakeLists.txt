cmake_minimum_required(VERSION 3.20)
project(daylog-ai LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(daylog-ai src/main.cpp)
if(MSVC)
  target_compile_options(daylog-ai PRIVATE /W4 /utf-8)
else()
  target_compile_options(daylog-ai PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS daylog-ai RUNTIME DESTINATION ai)
